import pandas as pd
import matplotlib.pyplot as plt
from database_connection import create_connection
from pathlib import Path


BASE_DIR = Path(__file__).resolve().parent.parent
OUTPUT_DIR = BASE_DIR / "outputs" / "charts"

OUTPUT_DIR.mkdir(
    parents=True,
    exist_ok=True
)


connection = create_connection()

query = """
SELECT
    m.machine_name,
    SUM(p.planned_quantity) AS planned_quantity,
    SUM(p.produced_quantity) AS produced_quantity,
    SUM(p.downtime_hours) AS downtime_hours
FROM production_records p
JOIN machines m
    ON p.machine_id = m.machine_id
GROUP BY m.machine_name
"""

data = pd.read_sql_query(query, connection)

connection.close()


data["efficiency"] = (
    data["produced_quantity"]
    / data["planned_quantity"]
    * 100
)


plt.figure(figsize=(10, 6))

plt.bar(
    data["machine_name"],
    data["efficiency"]
)

plt.title("Machine Production Efficiency")
plt.xlabel("Machine")
plt.ylabel("Efficiency (%)")

plt.tight_layout()

plt.savefig(
    OUTPUT_DIR / "machine_efficiency.png"
)

plt.close()


plt.figure(figsize=(10, 6))

plt.bar(
    data["machine_name"],
    data["downtime_hours"]
)

plt.title("Machine Downtime Analysis")
plt.xlabel("Machine")
plt.ylabel("Downtime (Hours)")

plt.tight_layout()

plt.savefig(
    OUTPUT_DIR / "machine_downtime.png"
)

plt.close()


print("Charts successfully generated.")
