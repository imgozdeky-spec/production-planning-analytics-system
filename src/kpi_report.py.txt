import pandas as pd
from database_connection import create_connection
from pathlib import Path


BASE_DIR = Path(__file__).resolve().parent.parent
REPORT_DIR = BASE_DIR / "outputs" / "reports"

REPORT_DIR.mkdir(
    parents=True,
    exist_ok=True
)


connection = create_connection()

query = """
SELECT
    COUNT(DISTINCT o.order_id) AS total_orders,
    SUM(o.quantity) AS total_order_quantity,
    COALESCE(SUM(p.produced_quantity), 0) AS total_produced_quantity,
    COALESCE(SUM(p.downtime_hours), 0) AS total_downtime_hours,
    COALESCE(SUM(p.production_time_hours), 0) AS total_production_hours
FROM orders o
LEFT JOIN production_records p
    ON o.order_id = p.order_id
"""

kpi = pd.read_sql_query(query, connection)

connection.close()


kpi["overall_completion_rate"] = (
    kpi["total_produced_quantity"]
    / kpi["total_order_quantity"]
    * 100
)

kpi["machine_utilization"] = (
    kpi["total_production_hours"]
    /
    (
        kpi["total_production_hours"]
        + kpi["total_downtime_hours"]
    )
    * 100
)

kpi = kpi.round(2)


print("Production Planning KPI Report")
print("--------------------------------")
print(kpi.T)


report_path = REPORT_DIR / "production_kpi_report.csv"

kpi.to_csv(
    report_path,
    index=False
)

print(f"\nReport saved to: {report_path}")
